# Run from anywhere: bash scriptDCVRPTW.txt
ROOT="$(cd "$(dirname "$0")" && pwd)"
APP="$ROOT/Applications/VRPTWDC"
mkdir -p "$ROOT/out/DCVRPTW"

for f in "$APP"/Solomon/*.txt; do
    filename=$(basename "$f" .txt)
    for dur in 360 480 600; do
        echo "Rodando $filename com duração $dur..."
        julia --project="$APP" "$APP/mainVRPTWDC.jl" "$filename.txt" "$dur" 2>&1 | tee "$ROOT/out/DCVRPTW/${filename}_${dur}.log"
    done
done
